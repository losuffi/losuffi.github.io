    MkDocs + GitHub Pages 周更技术博客实施方案（集成版）¶

    一、方案概述¶

    将 MkDocs 静态站点生成器集成到现有的 TaskManager 项目中，复用现有的 GitHub 同步功能，实现以周为单位组织和展示技术文档的博客系统。

    集成优势¶

    • ✅ 复用现有功能: 利用已有的 BlogWindow 和 GitHubSync 模块
    • ✅ 统一管理: Markdown 文件统一存储在 data/markdowns/ 目录
    • ✅ 自动同步: 通过现有的 GitHub 同步功能自动备份内容
    • ✅ 周刊展示: 通过 MkDocs 生成美观的周刊网站

    二、技术栈¶

    • MkDocs: 静态站点生成器
    • Material for MkDocs: 现代化主题（推荐）
    • GitHub Pages: 免费静态网站托管
    • GitHub Actions: 自动化部署
    • 现有 BlogWindow: Markdown 文件管理和编辑
    • 现有 GitHubSync: 文件同步到 GitHub

    5.1 创建新周刊脚本 (scripts/new_week.py)¶

    #!/usr/bin/env python3
"""
创建新周刊脚本
在 data/markdowns/weeks/ 目录下创建新的周刊文件夹
"""
import os
import datetime
from pathlib import Path

def get_week_info():
    """获取当前周信息"""
    today = datetime.date.today()
    year, week_num, _ = today.isocalendar()

    # 计算周的起止日期（周一到周日）
    start_of_week = today - datetime.timedelta(days=today.weekday())
    end_of_week = start_of_week + datetime.timedelta(days=6)

    return year, week_num, start_of_week, end_of_week

def create_week_folder():
    """创建周刊文件夹和模板文件"""
    year, week, start_date, end_date = get_week_info()

    # 创建周刊目录
    week_path = Path(f"data/markdowns/weeks/{year}/week-{week:02d}")
    week_path.mkdir(parents=True, exist_ok=True)

    # 创建 index.md 模板
    index_content = f"""# {year}年第{week}周

!!! info "时间范围"
    {start_date.strftime('%Y-%m-%d')} 至 {end_date.strftime('%Y-%m-%d')}

## 本周概览

本周主要分享了以下内容...（请补充）

## 文章列表

### [文章标题1](article-1.md)
文章简介...

**关键词**: 标签1, 标签2

---

## 导航

- [← 上一周](../week-{week-1:02d}/index.md)
- [下一周 →](../week-{week+1:02d}/index.md)
"""

    index_file = week_path / "index.md"
    if not index_file.exists():
        index_file.write_text(index_content, encoding='utf-8')
        print(f"✅ 已创建周刊概览: {index_file}")
    else:
        print(f"⚠️  文件已存在: {index_file}")

    # 创建示例文章
    article_template = f"""# 文章标题

!!! tip "发布日期"
    {start_date.strftime('%Y-%m-%d')}

## 简介

在这里写文章内容...

## 内容

...

---

[← 返回第{week}周概览](index.md)
"""

    article_file = week_path / "article-1.md"
    if not article_file.exists():
        article_file.write_text(article_template, encoding='utf-8')
        print(f"✅ 已创建示例文章: {article_file}")

    # 同步到 docs/ 目录（如果使用文件复制方式）
    sync_to_docs(week_path, year, week)

    print(f"\n🎉 第{week}周刊创建完成！")
    print(f"📂 目录: {week_path}")
    print(f"\n下一步:")
    print(f"1. 在 TaskManager Blog 模块中编辑文章")
    print(f"2. 使用 GitHub 同步功能上传到仓库")
    print(f"3. 更新 mkdocs.yml 中的导航（或运行 generate_nav.py 自动生成）")

def sync_to_docs(week_path, year, week):
    """同步周刊到 docs/ 目录"""
    docs_week_path = Path(f"docs/weeks/{year}/week-{week:02d}")
    docs_week_path.parent.mkdir(parents=True, exist_ok=True)

    # 在 Windows 上，如果没有管理员权限，使用文件复制
    # 在 Linux/Mac 上，可以使用符号链接
    import shutil
    if docs_week_path.exists():
        shutil.rmtree(docs_week_path)
    shutil.copytree(week_path, docs_week_path)
    print(f"✅ 已同步到 docs/ 目录")

if __name__ == "__main__":
    create_week_folder()

    5.2 自动生成导航脚本 (scripts/generate_nav.py)¶

    #!/usr/bin/env python3
"""
自动生成 MkDocs 导航配置
扫描 data/markdowns/weeks/ 目录，生成导航结构
"""
import os
import yaml
from pathlib import Path
from collections import defaultdict

def scan_weeks():
    """扫描所有周刊文件夹"""
    weeks_dir = Path("data/markdowns/weeks")

    if not weeks_dir.exists():
        print("⚠️  未找到 weeks 目录")
        return {}

    # 按年份和周数组织
    weeks_structure = defaultdict(list)

    for year_dir in sorted(weeks_dir.iterdir(), reverse=True):
        if not year_dir.is_dir():
            continue

        year = year_dir.name

        for week_dir in sorted(year_dir.iterdir(), reverse=True):
            if not week_dir.is_dir() or not week_dir.name.startswith('week-'):
                continue

            week_num = week_dir.name.replace('week-', '')
            index_file = week_dir / "index.md"

            if index_file.exists():
                weeks_structure[year].append({
                    'week': week_num,
                    'path': f"weeks/{year}/week-{week_num}/index.md"
                })

    return weeks_structure

def generate_nav_config(weeks_structure):
    """生成导航配置"""
    nav = [
        {'首页': 'index.md'}
    ]

    # 生成周刊导航
    if weeks_structure:
        weekly_nav = []

        for year in sorted(weeks_structure.keys(), reverse=True):
            year_weeks = []
            for week_info in weeks_structure[year]:
                week_num = week_info['week']
                week_label = f"第{week_num}周"
                year_weeks.append({week_label: week_info['path']})

            weekly_nav.append({f"{year}年": year_weeks})

        nav.append({'周刊': weekly_nav})

    nav.extend([
        {'归档': 'archive.md'},
        {'关于': 'about.md'}
    ])

    return nav

def update_mkdocs_config():
    """更新 mkdocs.yml 配置文件"""
    config_file = Path("mkdocs.yml")

    if not config_file.exists():
        print("❌ 未找到 mkdocs.yml 文件")
        return

    # 读取现有配置
    with open(config_file, 'r', encoding='utf-8') as f:
        config = yaml.safe_load(f)

    # 扫描周刊并生成导航
    weeks_structure = scan_weeks()
    new_nav = generate_nav_config(weeks_structure)

    # 更新导航配置
    config['nav'] = new_nav

    # 写回配置文件
    with open(config_file, 'w', encoding='utf-8') as f:
        yaml.dump(config, f, allow_unicode=True, sort_keys=False, default_flow_style=False)

    print("✅ 已更新 mkdocs.yml 导航配置")
    print(f"📊 找到 {sum(len(weeks) for weeks in weeks_structure.values())} 个周刊")

if __name__ == "__main__":
    update_mkdocs_config()

    5.3 同步脚本 (scripts/sync_to_mkdocs.py)¶

    #!/usr/bin/env python3
"""
同步 data/markdowns/ 到 docs/ 目录
解决 Windows 软链接权限问题
"""
import shutil
from pathlib import Path

def sync_markdowns():
    """同步 markdowns 目录到 docs"""
    source = Path("data/markdowns/weeks")
    target = Path("docs/weeks")

    if not source.exists():
        print("⚠️  源目录不存在:", source)
        return

    # 删除旧的目标目录
    if target.exists():
        shutil.rmtree(target)

    # 复制整个目录
    shutil.copytree(source, target)
    print(f"✅ 已同步 {source} -> {target}")

    # 统计文件数
    md_files = list(target.rglob("*.md"))
    print(f"📄 同步了 {len(md_files)} 个 Markdown 文件")

if __name__ == "__main__":
    sync_markdowns()

    name: Deploy MkDocs Blog to GitHub Pages

on:
  push:
    branches:
      - main  # 或 master

      - 'data/markdowns/**'
      - 'docs/**'

permissions:
  contents: write

jobs:
  deploy:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout repository
        uses: actions/checkout@v4

      - name: Setup Python
        uses: actions/setup-python@v5
        with:
          python-version: 3.x

      - name: Install dependencies
        run: |
          pip install mkdocs-material PyYAML

      - name: Sync markdowns to docs
        run: |
          python scripts/sync_to_mkdocs.py

      - name: Generate navigation
        run: |
          python scripts/generate_nav.py

      - name: Build and Deploy
        run: |
          mkdocs gh-deploy --force --clean --verbose

def create_new_week(self):
    """创建新周刊"""
    import subprocess
    try:
        # 调用脚本创建周刊
        result = subprocess.run(
            ['python', 'scripts/new_week.py'],
            capture_output=True,
            text=True
        )
        if result.returncode == 0:
            messagebox.showinfo(
                "创建成功",
                f"新周刊创建成功！\n\n{result.stdout}",
                parent=self.window
            )
            self.refresh_file_list()
        else:
            messagebox.showerror(
                "创建失败",
                f"创建周刊时出错：\n{result.stderr}",
                parent=self.window
            )
    except Exception as e:
        messagebox.showerror(
            "错误",
            f"无法创建周刊：{str(e)}",
            parent=self.window
        )

    6.2 修改 BlogWindow 的文件列表显示¶

    可以增强文件列表，按周刊分组显示：

    def refresh_file_list(self):
    """刷新文件列表（增强版：支持周刊分组）"""
    # 清空现有列表
    for item in self.file_tree.get_children():
        self.file_tree.delete(item)

    weeks_dir = os.path.join(self.markdowns_dir, "weeks")
    standalone_dir = os.path.join(self.markdowns_dir, "standalone")

    # 1. 添加周刊分组
    if os.path.exists(weeks_dir):
        weeks_node = self.file_tree.insert('', 'end', text="📅 周刊", open=True)

        # 遍历年份
        for year in sorted(os.listdir(weeks_dir), reverse=True):
            year_path = os.path.join(weeks_dir, year)
            if not os.path.isdir(year_path):
                continue

            year_node = self.file_tree.insert(weeks_node, 'end', text=f"{year}年", open=True)

            # 遍历周
            for week in sorted(os.listdir(year_path), reverse=True):
                week_path = os.path.join(year_path, week)
                if not os.path.isdir(week_path):
                    continue

                week_node = self.file_tree.insert(year_node, 'end', text=f"📖 {week}", open=False)

                # 遍历文章
                for filename in sorted(os.listdir(week_path)):
                    if filename.endswith('.md'):
                        file_path = os.path.join(week_path, filename)
                        display_name = filename[:-3]
                        self.file_tree.insert(
                            week_node, 'end',
                            text=f"  📄 {display_name}",
                            values=(file_path,)
                        )

    # 2. 添加独立文章分组
    if os.path.exists(standalone_dir):
        standalone_node = self.file_tree.insert('', 'end', text="📚 独立文章", open=True)

        for filename in sorted(os.listdir(standalone_dir)):
            if filename.endswith('.md'):
                file_path = os.path.join(standalone_dir, filename)
                display_name = filename[:-3]
                self.file_tree.insert(
                    standalone_node, 'end',
                    text=f"📄 {display_name}",
                    values=(file_path,)
                )

    八、常见问题与解决方案¶

    Q1: Windows 下无法创建符号链接¶

    解决: 使用 sync_to_mkdocs.py 脚本复制文件而不是创建符号链接。

    Q2: GitHub Actions 构建失败¶

    检查项: - Python 依赖是否正确安装（pip install mkdocs-material PyYAML） - mkdocs.yml 配置是否有语法错误 - docs/ 目录结构是否正确

    Q3: 博客页面显示 404¶

    解决: - 检查 GitHub Pages 设置是否选择了 gh-pages 分支 - 确认仓库是公开的 - 等待几分钟让 GitHub Pages 部署完成

    Q4: 同步后导航没有更新¶

    解决: 在推送到 GitHub 前，手动运行导航生成脚本： 

    python scripts/generate_nav.py
git add mkdocs.yml
git commit -m "Update navigation"
git push

    Q5: 中文文件名显示问题¶

    建议: - 文件名使用英文和连字符 - 中文标题写在 Markdown 的 H1 标题中 - 例如：efficient-computation.md → # 高效计算
